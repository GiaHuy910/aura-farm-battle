cho user chỉ số aura mặc định 50 đi
(đây là hệ số cố định nên được config)
user_base_aura = 100
-> khi này ông sẽ có chính là field 
      "base_stats": {
        "aura": 100
      },

hệ số tier căn bản cho từng tier (áp dụng clothes và skills)
(đây là hệ số cố định nên được config)
common_base = 0.3
uncommon_base = 0.6
rare_base = 1.2
epic_base = 2.2
legendary_base = 3.4
mythical_base = 5.0

Cách tính aura flat cho clothes
(đây là hệ số biến động không cần config và chỉ để làm ví dụ cho giai đoạn khởi tạo 1 cloth)
common_clothe_base = common_base * user_base_aura
uncommon_clothe_base = uncommon_base * user_base_aura
rare_clothe_base = rare_base * user_base_aura
epic_clothe_base = epic_base * user_base_aura
legendary_clothe_base = legendary_base * user_base_aura
mythical_clothe_base = mythical_base * user_base_aura
-> khi này ông sẽ có chính là field 
    "rarity":"epic",
    "bonus_stats": {
      "aura": 220 
    }

VÍ DỤ sau khi user đã trang bị xong 3 món epic epic epic 
(đây là hệ số biến động và chỉ để làm ví dụ cho cách tính total aura của user sau khi khởi tạo user)
user_total_aura = user_base_aura + epic_clothe_base + epic_clothe_base + epic_clothe_base
user_total_aura = 825
khi này ô sẽ có field 
      "final_stats": {
        "aura": 0
      }

hệ số của skills ( skill được thiết kế sẽ có chỉ số là % aura dựa trên total CURRENT aura 
của user - tức là sẽ dựa trên aura hiện tại TRONG TRẬN mà nhân tiếp, càng đánh càng mạnh )
(đây là hệ số cố định cần được config)
common_skill_multiplier = common_base (0.4 có nghĩa là 40%)
uncommon_skill_multiplier = uncommon_base 
rare_skill_multiplier = rare_base 
epic_skill_multiplier = epic_base
legendary_skill_multiplier = legendary_base
mythical_skill_multiplier = mythical_base
-> khi này ô sẽ common_skill_multiplier chính là field 
    "bonus_stats_proc": { "aura_mul": 2.2 }

đây là hệ số cố định cần config
ult_multiplier = 2.5

hệ số của ult thì khác chút, cơ bản là x2.5 so với skill thôi
(đây chỉ là hệ số làm ví dụ k cần config)
common_ult_multiplier = common_base * ult_multiplier
uncommon_ult_multiplier = uncommon_base * ult_multiplier 
rare_ult_multiplier = rare_base * ult_multiplier 
epic_ult_multiplier = epic_base * ult_multiplier
legendary_ult_multiplier = legendary_base * ult_multiplier 
mythical_ult_multiplier = mythical_base * ult_multiplier 
-> khi này ô sẽ có common_ult_multiplier chính là field    
  "bonus_stats_proc": { "aura_mul": 5.5 }
